  \begin{block}{Abstract}
    Wireless Sensor Networks~(WSNs) deployed in disaster-monitoring scenarios require energy-efficient, low-latency multi-hop routing under strict resource constraints.
    We propose \textbf{Graph-Attentive Policy Fusion~(GAPF)}, a lightweight meta-learning framework that fuses frozen MARL expert policies (QMIX, QTRAN) through a \textbf{Graph Convolutional Network~(GCN)} encoder and a \textbf{Contextual Algorithm Selection~(CAS)} controller.
    GAPF encodes the sensor topology into a compact graph embedding and selects the best-performing expert with only \textbf{1,473 trainable parameters}.
    Evaluated across 7~scenarios (20--100~sensors, 21,000 test episodes), GAPF achieves $\geq$98\,\% packet delivery, +43\,\% to +111\,\% improvement over baselines, 2--3.5$\times$ less energy, 3--9$\times$ better energy balance, and 13$\times$ less memory.

    \smallskip
    \textbf{Keywords:} WSN, MARL, GNN, policy fusion, energy-efficient routing, disaster monitoring.
  \end{block}

  \begin{block}{Introduction \& Motivation}
    \begin{itemize}
      \item Disaster-monitoring WSNs require \textbf{reliable, energy-efficient multi-hop routing} to a base station under strict resource constraints.
      \item Existing MARL approaches (QMIX, QTRAN) treat routing independently, ignoring the network topology, and suffer from \textbf{memory explosion at scale} (up to 13.8\,GB).
      \item Metaheuristic baselines (e.g.\ QPSOFL) see packet delivery \textbf{collapse below 10\,\%} in large networks.
      \item \textbf{Goal:} Design a lightweight, topology-aware meta-controller that selects the best routing expert per deployment while satisfying energy, latency, and memory constraints.
    \end{itemize}
  \end{block}

  \begin{block}{Problem Formulation}
    \textbf{Setting:} Base station~(BS) at the centre of a $100\!\times\!100$\,m field $+$ $N$ sensor nodes.
    Each sensor starts with $E_0=1$\,J and holds 1~packet.  Multi-hop relay to BS within radius~$R$.

    \heading{Energy Model}
    \begin{equation*}
    \left\{
    \begin{aligned}
      E_T(M,d) &= M \cdot a \cdot (E_{\text{elec}} + E_{\text{amp}} \cdot d^2) \\
      E_R(M) &= M \cdot a \cdot E_{\text{elec}}
    \end{aligned}
    \right.
    \end{equation*}
    with $E_{\text{elec}}=50$\,nJ/bit, $E_{\text{amp}}=100$\,pJ/(bit$\cdot$m$^2$), packet size $a=512$\,bits.

    \heading{Routing Objectives}
    \begin{enumerate}
      \item \textbf{Maximize} packet delivery ratio~(PDR).
      \item \textbf{Minimize} total energy consumption $\sum (E_T + E_R)$.
      \item \textbf{Balance} energy usage across sensors (minimize $\sigma(e_{\text{rem}})$).
    \end{enumerate}
  \end{block}

  \begin{block}{Reward Scalarization}
    4-dimensional reward vector per relay hop:
    \begin{equation*}
      \mathbf{r}_i^{(t)} = \bigl[\,\underbrace{r_1}_{\text{angle}},\;\underbrace{r_2}_{\text{energy}},\;\underbrace{r_3}_{\text{balance}},\;\underbrace{r_4}_{\text{load}}\,\bigr]
    \end{equation*}
    \textbf{Monotonic Q$\cdot$K Attention Network} scalarizes~$\mathbf{r}$:
    \begin{equation*}
      \alpha_j = \frac{\exp(\mathbf{k}_j^\top \mathbf{q}/\sqrt{d})}{\sum_{l=1}^{4}\exp(\mathbf{k}_l^\top \mathbf{q}/\sqrt{d})},
      \quad
      g_\psi(\mathbf{r}') = \mathbf{W}_3\,\sigma\!\bigl(\mathbf{W}_2\,\sigma\!\bigl(\mathbf{W}_1\,\mathbf{r}'+\mathbf{b}_1\bigr)+\mathbf{b}_2\bigr)+b_3
    \end{equation*}
    $\sigma(\cdot)=\text{softplus}$, all $\mathbf{W}_\ell\!\geq\!0$ (monotonicity), dims $4\!\to\!16\!\to\!16\!\to\!1$.

    \smallskip
    \textbf{Outer loop:} Evolution Strategies~(ES) tunes attention parameters to maximize
    $J = \text{energy-eff.} + \text{energy-balance} \in [0,2]$.
  \end{block}

\end{column}

  \begin{block}{GAPF Architecture}
    \begin{center}
    \begin{tikzpicture}[
        node distance=1.4cm and 1.8cm,
        >={Stealth[length=3mm]},
        block/.style={draw, rounded corners, minimum height=1.4cm,
                      minimum width=5.5cm, align=center, font=\small},
        expert/.style={draw, rounded corners, fill=blue!10,
                       minimum height=1.1cm, minimum width=3.8cm,
                       align=center, font=\small},
        env/.style={draw, rounded corners, fill=green!10,
                    minimum height=1.1cm, minimum width=5cm,
                    align=center, font=\small},
        arrow/.style={->, thick}
      ]
      % ---------- nodes ----------
      \node[env] (graph)
        {Sensor Graph $\mathcal{G}_k = (\mathcal{V},\mathcal{E}_k)$\\
         $\mathbf{x}_i\!\in\!\mathbb{R}^4$ per node};

      \node[block, below=of graph, fill=orange!15] (gcn)
        {\textbf{GCN Encoder} (2-layer, $d\!=\!16$)\\
         $\hat{\mathbf{A}}\mathbf{X}\mathbf{W}_1
          \!\to\!\text{ReLU}
          \!\to\!\hat{\mathbf{A}}\mathbf{H}\mathbf{W}_2$\\
         Mean-pool $\to \mathbf{z}\!\in\!\mathbb{R}^{16}$};

      \node[block, below=of gcn, fill=yellow!15] (cas)
        {\textbf{CAS Controller} (MLP $16\!\to\!64\!\to\!1$)\\
         $p_{\text{QMIX}}
          =\sigma(\mathbf{w}_2^\top
          \text{ReLU}(\mathbf{W}_c\mathbf{z}+\mathbf{b}_c)+b_2)$};

      \node[expert, below left=1.8cm and 0.8cm of cas]  (qmix)  {Frozen \textbf{QMIX}};
      \node[expert, below right=1.8cm and 0.8cm of cas] (qtran) {Frozen \textbf{QTRAN}};

      \node[env, below=4cm of cas] (wsn)
        {\textbf{WSN Environment}\\Per-agent GRU policy};

      % ---------- arrows ----------
      \draw[arrow] (graph) -- (gcn);
      \draw[arrow] (gcn)   -- (cas);
      \draw[arrow] (cas)   -| node[above left,  font=\footnotesize]{select} (qmix);
      \draw[arrow] (cas)   -| node[above right, font=\footnotesize]{or}     (qtran);
      \draw[arrow] (qmix)  |- (wsn);
      \draw[arrow] (qtran) |- (wsn);
      \draw[arrow, dashed] (wsn.east) -- ++(2.5,0)
            |- node[right, font=\footnotesize, pos=0.25]{$G_k$ (return)} (graph.east);
    \end{tikzpicture}
    \end{center}

    \heading{Trainable Parameters}
    \begin{equation*}
      |\theta|
      = \underbrace{4\!\times\!16 + 16\!\times\!16}_{\text{GCN: }320}
      \;+\;
      \underbrace{16\!\times\!64 + 64 + 64\!\times\!1 + 1}_{\text{CAS: }1{,}153}
      = \mathbf{1{,}473}
    \end{equation*}
    $\sim$26$\times$ smaller than a single RNN agent (38,983~params).
    Trained via \textbf{REINFORCE} with EMA baseline ($\beta\!=\!0.95$).
  \end{block}

  \begin{block}{Design Rationale}
    \begin{itemize}
      \item \textbf{Frozen experts} --- avoids catastrophic forgetting; only the meta-controller is trained.
      \item \textbf{GCN encoder} --- captures topology (node degree, proximity to~BS); critical in sparse networks.
      \item \textbf{Episode-level selection} --- one forward pass of GCN+CAS per episode ($\sim$0.03\,ms).
      \item \textbf{CTDE deployment} --- GCN+CAS on gateway (6\,KB); each sensor runs a GRU ($\sim$100\,KB flash, $\sim$2\,KB SRAM).
      \item \textbf{ES outer loop} --- derivative-free tuning of reward attention; robust to non-differentiable objectives.
    \end{itemize}
  \end{block}

  \begin{block}{Key Findings}
    \begin{description}
      \item[\textbf{F1}] GAPF $\geq$98\,\% PDR; advantage grows in sparser networks (+111\,\% at $n\!=\!100$, $R\!=\!35$).
      \item[\textbf{F2}] 2--3.5$\times$ lower energy, 3--9$\times$ better balance vs QMIX/QTRAN.
      \item[\textbf{F3}] QMIX\,$\approx$\,QTRAN ($<$2\,\% gap): gain comes from \emph{topology-aware fusion}, not expert choice alone.
      \item[\textbf{F4}] QPSOFL collapses at scale; its low energy is an artifact of near-zero delivery.
      \item[\textbf{F5}] GAPF is the \textbf{only algorithm passing all feasibility constraints} (memory, latency).
    \end{description}
  \end{block}

  \begin{block}{Conclusion}
    \textbf{GAPF} demonstrates that lightweight policy fusion with graph-structural awareness can replace heavyweight monolithic MARL for resource-constrained, safety-critical WSN routing.
    With only 1,473 parameters it achieves near-perfect delivery, 2--3.5$\times$ less energy, and feasible deployment on embedded hardware (ARM Cortex-M4, $\sim$100\,KB flash).

    \smallskip
    \textbf{Future work:} hardware-in-the-loop validation on nRF52840 motes; cross-domain disaster monitoring (flood/wildfire/earthquake); dynamic step-level expert fusion.
  \end{block}
