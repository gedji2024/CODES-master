  \begin{block}{Abstract}
    Wireless Sensor Networks~(WSNs) deployed for disaster monitoring face three challenges: (i)~limited energy, (ii)~full-lifecycle carbon footprint, and (iii)~vulnerability of AI-learned policies to adversarial attacks.
    This thesis proposes three complementary frameworks forming a \textbf{complete protocol stack}:

    \smallskip
    \textbf{GAPF} --- lightweight MARL policy fusion via GCN (\textbf{1,473 params}, PDR~$\geq$~98\%, 2--3.5$\times$ less energy);\\
    \textbf{CALASH} --- carbon-aware 4-pillar protocol with Lyapunov guarantees (LCI~$-$33\%, lifetime~+60\%);\\
    \textbf{CASTER-ZT} --- zero-trust shielded decision control (0\%~FP, 3.07\,ms/decision, 10~formal propositions).

    \smallskip
    \textbf{Keywords:} WSN, MARL, GNN, policy fusion, lifecycle carbon, zero-trust, 6G, disaster monitoring.
  \end{block}

  \begin{block}{Motivation \& Research Questions}
    \begin{itemize}
      \item \textbf{7,348} major disasters (2000--2019); climate change increases frequency +30\%.
      \item WSNs: non-rechargeable batteries, limited RAM (16--256\,KB), physical vulnerability.
      \item Existing solutions address at most \textbf{4/12 design criteria} individually.
    \end{itemize}

    \heading{Three Specific Research Questions}
    \begin{description}
      \item[\textbf{Q1}] Can a lightweight graph-neural-network controller fuse pre-trained MARL policies to achieve a packet delivery ratio above 98\% with fewer than 2,000 trainable parameters?
      \item[\textbf{Q2}] To what extent can a four-pillar carbon-aware protocol, combined with 6G dual-band physical-layer capabilities, reduce the lifecycle carbon intensity of a disaster-monitoring WSN?
      \item[\textbf{Q3}] Is a zero-trust decision shield able to detect rogue AI policies with zero false positives and enforce graduated responses within the 10\,ms O-RAN real-time budget?
    \end{description}
  \end{block}

    \begin{center}

        node distance=1.2cm and 1.4cm,
        >={Stealth[length=2.5mm]},

      \node[block, below=of graph, fill=orange!15] (gcn)
        {\textbf{GCN Encoder} (2-layer, $d\!=\!16$)\\
         $\hat{\mathbf{A}}\mathbf{X}\mathbf{W}_1

          \!\to\!\hat{\mathbf{A}}\mathbf{H}\mathbf{W}_2$\\
         Mean-pool $\to \mathbf{z}\!\in\!\mathbb{R}^{16}$};

      \node[block, below=of gcn, fill=yellow!15] (cas)
        {\textbf{CAS Controller} (MLP $16\!\to\!64\!\to\!1$)\\
         $p_{\text{QMIX}}
          =\sigma(\mathbf{w}_2^\top
          \text{ReLU}(\mathbf{W}_c\mathbf{z}+\mathbf{b}_c)+b_2)$};

      \node[expert, below left=1.5cm and 0.5cm of cas]  (qmix)  {Frozen \textbf{QMIX}};
      \node[expert, below right=1.5cm and 0.5cm of cas] (qtran) {Frozen \textbf{QTRAN}};

      \draw[arrow] (graph) -- (gcn);
      \draw[arrow] (gcn)   -- (cas);
      \draw[arrow] (cas)   -| node[above left,  font=\footnotesize]{select} (qmix);
      \draw[arrow] (cas)   -| node[above right, font=\footnotesize]{or}     (qtran);
    \end{tikzpicture}
    \end{center}

    \heading{Trainable Parameters}
    \vspace{-6pt}
    \begin{equation*}
      |\theta|
      = \underbrace{4\!\times\!16 + 16\!\times\!16}_{\text{GCN: }320}
      \;+\;
      \underbrace{16\!\times\!64 + 64 + 64\!\times\!1 + 1}_{\text{CAS: }1{,}153}
      = \mathbf{1{,}473}
    \end{equation*}

    \heading{Bi-Level Training}
    \begin{itemize}
      \item \textbf{Inner loop:} MARL experts (QMIX/QTRAN) via EPyMARL; $\alpha\!=\!3\!\times\!10^{-4}$, $\epsilon$: $0.5\!\to\!0.01$ (150K steps), replay 15K.
      \item \textbf{Outer loop:} ES tunes Q$\cdot$K attention ($\sigma_{\text{ES}}\!=\!0.01$); REINFORCE trains GCN+CAS ($\alpha\!=\!3\!\times\!10^{-3}$, $\beta\!=\!0.95$).
    \end{itemize}
  \end{block}
