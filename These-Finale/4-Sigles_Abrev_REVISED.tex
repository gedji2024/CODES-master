\noindent
Les symboles ci-dessous sont regroupés par thème et listés par ordre d'apparition au sein de chaque groupe. L'unité est précisée entre crochets lorsqu'elle est applicable ; ``sans dim.'' indique une grandeur adimensionnelle.

\bigskip

\paragraph{Variables d'état du réseau}
\begin{longtable}{lp{3in}l}
$\mathcal{G}(t)=(\mathcal{V},\mathcal{E}(t))$ & Graphe du réseau WSN à l'instant $t$ & --- \\
$\mathcal{V}$ & Ensemble des nœuds capteurs & --- \\
$N$ & Nombre total de nœuds & [nœuds] \\
$\mathcal{E}(t)$ & Ensemble des liens actifs à l'instant $t$ & --- \\
$d_{ij}$ & Distance entre nœuds $i$ et $j$ & [m] \\
$d_0$ & Distance seuil de transition propagation & [m] \\
$e_i(t)$ & Énergie résiduelle du nœud $i$ à l'instant $t$ & [J] \\
$E_0$ & Énergie initiale de chaque nœud & [J] \\
$\mathcal{N}(i)$ & Ensemble des voisins à 1 saut du nœud $i$ & --- \\
\end{longtable}

\paragraph{Modèle énergétique (Heinzelman~\emph{et al.})}
\begin{longtable}{lp{3in}l}
$E_{\text{elec}}$ & Énergie dissipée par bit d'émission/réception & [J/bit] \\
$\varepsilon_{\text{fs}}$ & Coefficient d'amplification espace libre & [J/bit/m\textsuperscript{2}] \\
$\varepsilon_{\text{mp}}$ & Coefficient d'amplification multi-chemins & [J/bit/m\textsuperscript{4}] \\
$\ell$ & Taille du paquet transmis & [bits] \\
$\alpha$ & Exposant de perte de chemin (2 ou 4) & [sans dim.] \\
$E_{\text{Tx}}(\ell, d)$ & Énergie de transmission de $\ell$ bits sur distance $d$ & [J] \\
$E_{\text{Rx}}(\ell)$ & Énergie de réception de $\ell$ bits & [J] \\
\end{longtable}

\paragraph{Variables de l'apprentissage par renforcement multi-agents (MARL)}
\begin{longtable}{lp{3in}l}
$s_i^t$ & État local de l'agent $i$ à l'instant $t$ & --- \\
$o_i^t$ & Observation partielle de l'agent $i$ & --- \\
$a_i^t$ & Action de l'agent $i$ à l'instant $t$ & --- \\
$r_i^t$ & Récompense individuelle de l'agent $i$ & [sans dim.] \\
$\gamma$ & Facteur d'actualisation des récompenses futures & [sans dim.] \\
$\pi_\theta$ & Politique paramétrée par $\theta$ & --- \\
$Q^\pi(s,a)$ & Fonction valeur action-état sous la politique $\pi$ & [sans dim.] \\
$Q_{\text{tot}}$ & Valeur globale mélangée (QMIX/QTRAN) & [sans dim.] \\
$\mathbf{h}_i^{(k)}$ & Représentation GNN du nœud $i$ à la couche $k$ & --- \\
$\mathbf{z}_G$ & Représentation de niveau graphe (pooling) & --- \\
$w_j$ & Poids contextuel de la politique $j$ (CAS) & [sans dim.] \\
$|\Theta|$ & Nombre de paramètres entraînables & [paramètres] \\
$\boldsymbol{\alpha}$ & Vecteur de poids d'attention appris (GAPF outer loop) & [sans dim.] \\
$\phi$ & Paramètres du réseau d'attention (Evolution Strategies) & --- \\
$J(\phi)$ & Méta-objectif outer loop ES (efficacité + équilibre) & [sans dim.] \\
$\sigma_{\text{ES}}$ & Bruit de perturbation Evolution Strategies ($= 0.01$) & [sans dim.] \\
$\eta_{\text{ES}}$ & Taux d'apprentissage Evolution Strategies ($= 0.01$) & [sans dim.] \\
\end{longtable}

\paragraph{Métriques de performance réseau}
\begin{longtable}{lp{3in}l}
PDR & Taux de livraison de paquets (\emph{Packet Delivery Ratio}) & [\%] \\
$T_{\text{vie}}$ & Durée de vie du réseau (dernier nœud vivant) & [rounds] \\
$T_{1/2}$ & Demi-vie du réseau (50\% des nœuds vivants) & [rounds] \\
$\bar{e}(t)$ & Énergie résiduelle moyenne du réseau à $t$ & [J] \\
$\Delta_e$ & Variance d'équilibre énergétique entre nœuds & [J\textsuperscript{2}] \\
\end{longtable}

\paragraph{Variables carbone (CALASH)}
\begin{longtable}{lp{3in}l}
LCI & Intensité carbone du cycle de vie (\emph{Lifecycle Carbon Intensity}) & [gCO\textsubscript{2}eq/paquet] \\
$C_{\text{emb}}$ & Carbone incorporé (fabrication + installation) & [gCO\textsubscript{2}eq] \\
$C_{\text{op}}(t)$ & Carbone opérationnel cumulé jusqu'à $t$ & [gCO\textsubscript{2}eq] \\
$C_{\text{eol}}$ & Carbone de fin de vie (traitement DEEE) & [gCO\textsubscript{2}eq] \\
CI$(t)$ & Intensité carbone du réseau électrique à $t$ & [gCO\textsubscript{2}eq/kWh] \\
$\rho_{\text{CS}}(t)$ & Ratio de compression sensing adaptatif & [sans dim.] \\
$\rho_{\min}$ & Compression maximale (fraction minimale transmise, $= 0.3$) & [sans dim.] \\
$\rho_{\max}$ & Compression minimale (fraction maximale transmise, $= 0.8$) & [sans dim.] \\
$r_{\text{recycle}}$ & Taux de recyclage des nœuds morts ($= 0.3$) & [sans dim.] \\
$Z(t)$ & File virtuelle de Lyapunov (pression carbone) & [gCO\textsubscript{2}eq] \\
$\bar{c}$ & Budget carbone moyen admissible par ronde & [gCO\textsubscript{2}eq/ronde] \\
$V$ & Paramètre de compromis Lyapunov (livraison vs.~carbone) & [sans dim.] \\
$B$ & Constante de borne Lyapunov ($\frac{1}{2}\max_t\{(C_{\text{tot}}(t)-\bar{c})^2\}$) & [gCO\textsubscript{2}eq\textsuperscript{2}] \\
$\bar{D}$ & Taux de livraison moyen atteint par CALASH & [paquets/ronde] \\
$\bar{D}^*$ & Taux de livraison optimal de toute politique statique & [paquets/ronde] \\
$\ell_j$ & Facteur de pénalité carbone incorporé du nœud $j$ & [sans dim.] \\
$w_{\text{eol}}$ & Poids de la pénalité fin de vie dans la décision de routage & [sans dim.] \\
CDL & Borne de Pareto carbone-livraison-durée de vie & --- \\
\end{longtable}

\paragraph{Variables de détection comprimée / TinyML (CALASH)}
\begin{longtable}{lp{3in}l}
$n$ & Dimension originale du signal capteur & [sans dim.] \\
$m$ & Dimension compressée ($m = \rho_{\text{CS}} \cdot n$) & [sans dim.] \\
$s$ & Niveau de sparsité du signal dans le domaine DCT & [sans dim.] \\
$F$ & Fidélité de reconstruction ($F = 1 - \text{NMSE}$) & [sans dim.] \\
$\delta$ & Seuil de distorsion NMSE acceptable & [sans dim.] \\
\end{longtable}

\paragraph{Variables de sécurité (CASTER-ZT)}
\begin{longtable}{lp{3in}l}
$\mathcal{T}$ & Ensemble des types de menaces modélisées & --- \\
$p(y \mid o)$ & Distribution de confiance avec MC-Dropout & [sans dim.] \\
ECE & Erreur de calibration attendue (\emph{Expected Calibration Error}) & [sans dim.] \\
$\omega_{\text{rec}}$ & Score de récupération normalisé & [sans dim.] \\
FPR & Taux de faux positifs (\emph{False Positive Rate}) & [\%] \\
$\mathcal{D}$ & Bouclier déterministe à 5 niveaux gradués & --- \\
$\tau_k$ & Seuil du niveau $k$ du bouclier & [sans dim.] \\
RogueDet & Taux de détection d'agents rogues (nœuds compromis) & [\%] \\
$\eta_{\text{AI}}$ & Indice d'autonomie IA (étages neuronaux / total) & [sans dim.] \\
\end{longtable}
