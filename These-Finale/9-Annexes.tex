\Annexe{Preuves des propriétés théoriques de CASTER-ZT (matériel supplémentaire de l'Article~3)}\label{ann:proofs}\label{ann:caster-suppl}

Cette annexe reproduit les preuves complètes des dix propositions formelles énoncées dans l'Article~3 (Chapitre~\ref{sec:Theme3}). Ce matériel a été soumis en tant que document supplémentaire accompagnant l'article principal à IEEE Transactions on Mobile Computing. Les numéros de propositions correspondent à ceux du texte principal.

\medskip

\begin{proof}[Preuve de la Proposition~1 (Conservatisme monotone)]
Puisque $w_1, w_2, w_3, w_4 > 0$, le score composite de seuil $g_t = w_1(1-\tau_t)+w_2\rho_t+w_3 u_t+w_4\Delta_t$ est monotoniquement décroissant en $\tau_t$ et croissant en $\rho_t, u_t, \Delta_t$.
Puisque $\kappa_1, \kappa_2 \ge 0$, $\tau_{\min} = \tau_0 + \kappa_1\rho_t + \kappa_2 u_t$ est non décroissant en $\rho_t, u_t$, rendant la condition de confiance plus difficile à satisfaire.
La branche \textsc{allow} requiert $g_t < \gamma_1$, $\tau_t \ge \tau_{\min}$, et $\Delta_t \le \delta_{\max}$.
La dégradation de toute entrée ne peut qu'augmenter $g_t$, élever $\tau_{\min}$, ou violer la borne de divergence, poussant la sortie vers une branche plus conservatrice.
\end{proof}

\begin{proof}[Preuve de la Proposition~2 (Exclusion d'actuation non autorisée)]
Dans l'algorithme du bouclier, $\alpha_t=0$ déclenche $d_t=\textsc{block}$ à la vérification d'autorisation, avant que toute branche permissive ne soit évaluée.
\end{proof}

\begin{proof}[Preuve de la Proposition~3 (Sauvegarde de divergence)]
$\Delta_t > \delta_{\max}^{\mathrm{hard}}$ déclenche $d_t=\textsc{block}$ à la vérification de divergence dure.
$\Delta_t > \delta_{\max}$ viole la condition \textsc{allow}, empêchant les décisions permissives.
\end{proof}

\begin{proof}[Preuve de la Proposition~4 (Complétude décisionnelle bornée)]
L'algorithme du bouclier est une structure de branchement ordonnée finie avec des branches mutuellement exclusives (ordonnancement strict des seuils) et une clause \textsc{else} finale capturant tous les cas restants.
Toute entrée valide est donc associée à exactement un $d_t \in \mathcal{D}$.
\end{proof}

\begin{proof}[Preuve de la Proposition~5 (Complexité par décision)]
Encodage du graphe~: $\mathcal{O}(L|E_t|d)$.
Évaluation des candidats et évaluation confiance-risque~: $\mathcal{O}(|\mathcal{A}|k)$ chacune.
Évaluation du bouclier~: $\mathcal{O}(1)$.
Total~: $\mathcal{O}(L|E_t|d + |\mathcal{A}|k)$.
\end{proof}

\begin{proof}[Preuve de la Proposition~6 (Borne de défense en profondeur)]
Sous attaque d'identité, un \textsc{allow} non sûr requiert un faux négatif d'autorisation~: $\Pr \le \varepsilon_\alpha$.
Sous attaque de télémétrie~: $\Pr \le \varepsilon_\tau$.
Sous attaque combinée avec indépendance conditionnelle~:
$\Pr[\textsc{allow} \mid \text{combinée}] = \Pr[\alpha_t=1 \mid \text{id}] \cdot \Pr[g_t<\gamma_1 \mid \text{telem}] \le \varepsilon_\alpha \varepsilon_\tau$.
\end{proof}

\begin{proof}[Preuve de la Proposition~7 (Précision cohérente avec la calibration)]
Chaque épisode produit un Bernoulli $y_i$ avec $\mathbb{E}[y_i]=\varepsilon^\star$.
Par l'inégalité de Hoeffding~: $\Pr[\varepsilon^\star - \hat\varepsilon > t] \le \exp(-2nt^2)$.
En posant $\exp(-2nt^2)=\delta$, on obtient $t=\sqrt{\ln(1/\delta)/(2n)}$.
\end{proof}

\begin{proof}[Preuve de la Proposition~8 (Prix borné de la sécurité)]
Avec probabilité $(1-\eta-\eta_{\mathrm{block}})$, le bouclier autorise (perte~0).
Avec probabilité $\eta$, la réduction de portée engendre une perte $\le U_{\mathrm{gap}}$.
Avec probabilité $\eta_{\mathrm{block}}$, le blocage engendre une perte $\le U_{\max}$.
En sommant~: $\mathbb{E}[\text{perte}] \le \eta U_{\mathrm{gap}} + \eta_{\mathrm{block}} U_{\max}$.
\end{proof}

\begin{proof}[Preuve de la Proposition~9 (Couverture d'admissibilité conforme)]
Par le cadre conforme~\cite{angelopoulos2023conformal}, si les scores $(r_1,\ldots,r_{n+1})$ sont échangeables, $\Pr[r_{n+1} \le r_{(\lceil(1-\alpha)(n+1)\rceil)}] \ge 1-\alpha$.
Le bouclier autorise seulement si $g_t < \gamma_1$, ce qui correspond à $r_{n+1} > r_{(\lceil(1-\alpha)(n+1)\rceil)}$, se produisant avec probabilité $\le \alpha$.
\end{proof}

\begin{proof}[Preuve de la Proposition~10 (Taux de convergence de la calibration)]
Par la concentration de Hoeffding, $|\hat\varepsilon(\gamma) - \varepsilon(\gamma)| \le \sqrt{\ln(2/\delta)/(2n)}$ avec haute probabilité.
Puisque $\varepsilon'(\gamma) \ge c_\varepsilon > 0$, le théorème des accroissements finis donne $|\varepsilon(\gamma_1^{(n)}) - \varepsilon(\gamma_1^\star)| \ge c_\varepsilon |\gamma_1^{(n)} - \gamma_1^\star|$.
L'inégalité triangulaire donne $c_\varepsilon |\gamma_1^{(n)} - \gamma_1^\star| = \mathcal{O}(1/\sqrt{n})$, d'où $|\gamma_1^{(n)} - \gamma_1^\star| = \mathcal{O}_P(1/\sqrt{n})$.
\end{proof}

\section*{E.2~~Garantie de monotonie du réseau d'attention Q$\cdot$K}

\begin{theorem}[Garantie de monotonie]
Pour tout $\mathbf{r}^{(1)}, \mathbf{r}^{(2)} \in [0,1]^4$ avec $\mathbf{r}^{(1)} \leq \mathbf{r}^{(2)}$ composante par composante, la sortie du réseau satisfait $f_\phi(\mathbf{r}^{(1)}) \leq f_\phi(\mathbf{r}^{(2)})$.
\end{theorem}

\begin{proof}
Puisque $\alpha_i > 0$ pour tout $i$ (les poids d'attention sont obtenus par softmax), $\mathbf{r}^{(1)} \leq \mathbf{r}^{(2)}$ implique $\boldsymbol{\alpha} \odot \mathbf{r}^{(1)} \leq \boldsymbol{\alpha} \odot \mathbf{r}^{(2)}$.
Pour chaque couche monotone, le Jacobien satisfait $\partial g / \partial r'_i = \sigma'(\cdot) \cdot W_i \geq 0$, puisque la dérivée du softplus $\sigma'(x) = (1 + e^{-x})^{-1} \in (0,1)$ et $W_i \geq 0$ (les matrices de poids sont contraintes non négatives via $\mathbf{W}_\ell = \mathrm{softplus}(\mathbf{W}_\ell^{\mathrm{raw}})$).
Par la règle de la chaîne appliquée sur toutes les couches, la composition $f_\phi = g_\psi \circ (\boldsymbol{\alpha} \odot \cdot)$ est monotoniquement non décroissante en chaque composante d'entrée.
\end{proof}

\noindent Cette propriété garantit qu'une action Pareto-dominante (améliorant chaque objectif simultanément) reçoit toujours une récompense scalaire supérieure, prévenant toute mise en forme pathologique de la récompense.

\section*{F.1~~Preuve du Théorème CDL (Pareto carbon-delivery)}

\begin{proof}[Preuve du Théorème~\ref{art2:thm:cdl}]
Considérons la fonction de Lyapunov $L(t) = \frac{1}{2}Z(t)^2$ où $Z(t)$ est la file d'attente virtuelle de carbone.
La dérive conditionnelle à un pas satisfait~:
\[
\Delta(t) = \mathbb{E}[L(t+1) - L(t) \mid Z(t)] \le B + Z(t)\bigl(C_{\text{tot}}(t) - \bar{c}\bigr)
\]
où $B = \frac{1}{2}(C_{\max}^2 + \bar{c}^2)$ est une constante finie.

En ajoutant la pénalité $V \cdot (-D(t))$ et en notant que CALASH minimise goulûment le membre de droite à chaque saut via l'Éq.~\eqref{art2:eq:lyapunov_decision} avec le facteur lifecycle $\ell_j$, la borne dérive-plus-pénalité devient~:
\[
\Delta(t) - V \cdot D(t) \le B - V \cdot D^*(t)
\]

En télescopant sur $T$ rondes et utilisant $L(T) \ge 0$, on obtient~:
\[
\bar{D} \ge \bar{D}^* - B/V \quad \text{(livraison quasi-optimale)}
\]

La stabilité de la file ($\bar{Z} = O(V)$) donne~:
\[
\bar{C}_{\text{tot}} \le \bar{c} + B/V \quad \text{(budget carbone lifecycle)}
\]

Le compromis $V$ contrôle l'échange livraison--carbone~: un $V$ grand favorise la livraison au prix d'un dépassement transitoire du budget carbone.
\end{proof}
